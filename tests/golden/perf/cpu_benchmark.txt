# CPU benchmark budget (issue #241; measured numbers and machine in docs/design/performance.md
# section 9.1). This is an order-of-magnitude regression detector for the CI runners, not a speed
# ranking: every time budget is a generous multiple of the reference measurement, and the counters
# carry headroom too. Times are compared on the median of `runs` measured runs after `warmup_runs`.
#
# Raise a budget only with a recorded decision (a comment on the issue or an ADR), the same rule as
# tests/golden/memory/.

nodes = 12000
runs = 7
warmup_runs = 2

# Full paint (issue #257): a full repaint culls to the viewport, so it measures the visible part of
# the tree (5.7 ms, 232 nodes, against 87 ms and 12 000 before); 40 ms is about 7x, and
# `max_full_paint_nodes` below is the deterministic guard.
#
# Wall-clock budgets in integer milliseconds; see the table the test prints on every run. They are
# about six times the reference machine's median, which leaves roughly a factor of two on top of a
# presumed 3x-slower two-core gating runner: a real order-of-magnitude regression still fails, while
# CI noise and a loaded shared machine cannot.
max_first_style_layout_ms = 4500
max_full_paint_ms = 40
max_layout_paint_text_ms = 1800
max_layout_paint_container_ms = 1800
# Glass scenario (issue #244, glass-v1 section 9): one full paint of the reference scene with the
# fixed top bar blurred, recording the region copy, both blur passes and the composite on the null
# RHI. Same ~6x headroom as the other times: the composite adds a bounded number of commands, and
# the GPU blur itself is not measured here (see docs/design/performance.md section 9.1).
max_glass_paint_ms = 500
# Damaged scenario (issue #474): paint of a one-node edit through `begin_damaged`, with the CPU
# culling testing the plan's damage bounding box. Same ~6x time headroom; the quad count is a
# deterministic counter with small headroom, and it is what catches a culling regression (the
# whole-target count is `max_quads`).
max_damaged_paint_ms = 500
# The one-node edit reaches 2 quads; without the paint pass's damage clip it would be ~33, so 8 is
# a regression detector, not a tight bound.
max_damaged_quads = 8
# Nodes that resolve a style in the damaged paint (issue #487, docs/design/performance.md section
# 9.1). Deterministic: the culled paint is 5 on the reference scene, 11721 nodes are culled and 2
# clip-bounded subtrees are skipped whole. The whole tree is 12000, so this fails if the traversal
# regresses to painting everything (cull_paint off measures exactly 12000). Issue #490 took the count
# from 9353 to 5: layout now proves that a plain-text payload stays inside its content box, so the
# 78% of the tree that is text leaves is culled too, and only the damaged node and its ancestors
# paint. The budget is 6 - the smallest integer with headroom above 5 - so a platform difference in
# the culled-node set cannot fail the gate while a trip back to the whole tree still does.
max_damaged_paint_nodes = 6
# Nodes that resolve a style in the full paint (issue #257): 232 of 12000 on the reference scene,
# the visible part. 400 leaves headroom for a platform difference in the culled set, and a trip back
# to the whole tree (12000) still fails.
max_full_paint_nodes = 400

# Redraw and batching counters from the full paint. These are deterministic on every run, so they
# are compared with a small headroom rather than a factor; a changed count means the draw-list or
# batching story changed.
# Exactly one glass node in the glass scenario: a frame that stops compositing it is a regression.
glass_backdrops = 1
max_damage_rects = 8
max_draw_calls = 64
max_quads = 128
max_primitives = 12000
