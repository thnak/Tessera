# ThorVG spike: renders the representative icon/chart path set through the PathRasterizer seam and
# compares it against a checked-in golden. Only built when TESSERA_ENABLE_THORVG=ON.

add_executable(thorvg_spike
    main.cpp
    scene.cpp)

target_link_libraries(thorvg_spike PRIVATE
    tessera_path
    tessera_golden
    tessera_stb)

target_compile_features(thorvg_spike PRIVATE cxx_std_23)

if (TESSERA_BUILD_TESTS)
    # The tool subdirectory is added before the root enable_testing() call, so enable it here;
    # ctest still reads the top-level CTestTestfile generated by the root.
    enable_testing()
    add_test(
        NAME thorvg_spike_golden
        COMMAND thorvg_spike --compare ${PROJECT_SOURCE_DIR}/tests/golden/thorvg/icons.png)
endif()
