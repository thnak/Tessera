# Native Metal backend (ADR-053). No MoltenVK: this is the Metal API directly, in Objective-C++
# confined to this directory.
#
# Two halves, the same split the WebGPU backend uses:
#   - the API-free mapping tables and the MSL shader registry, built everywhere (cheap, pure) so
#     their unit tests run in the ordinary host ctest on Windows and Linux;
#   - the device half, which needs <Metal/Metal.h> and therefore only builds on Apple platforms.
#
# Tier 2 in-tree (ADR-049 section 4, docs/guides/writing-a-backend.md section 2) until the backend
# conformance suite passes on a macOS lane. The option is OFF by default and in no preset.

option(TESSERA_BUILD_METAL "Build the native Metal backend (Apple platforms only)" OFF)

# MSL sources (ADR-053 decision 2). Metal consumes MSL, not SPIR-V, so the shader table is built
# from ADR-049's cross-compile path: GLSL -> portable SPIR-V -> MSL via SPIRV-Cross, entirely inside
# the build directory, embedded as a string so production compiles no shader at run time.
#
# The MSL is embedded whenever the cross step produced it, which needs TESSERA_SHADER_CROSS=ON. A
# Metal build without it has an empty table and a pipeline that asks for a shader id fails loudly
# instead of drawing nothing, so the backend README makes turning it on part of the build recipe.
#
# The embedding is a build-time step, not a configure-time one: the MSL files are produced by the
# cross targets, which do not exist yet when this directory is configured. A configure-time read
# would therefore fail on a clean tree (it did), so each .inc is generated by a custom command that
# depends on the cross target.
set(TESSERA_METAL_MSL_DIR ${CMAKE_BINARY_DIR}/shaders/cross)
set(TESSERA_METAL_MSL_GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${TESSERA_METAL_MSL_GEN_DIR})

# Embeds one generated MSL file as a raw string literal at build time.
function(tessera_embed_msl shader_id stage out_var)
    set(source ${TESSERA_METAL_MSL_DIR}/${shader_id}/${shader_id}.${stage}.metal)
    set(output ${TESSERA_METAL_MSL_GEN_DIR}/${shader_id}_${stage}_msl.inc)
    add_custom_command(
        OUTPUT ${output}
        COMMAND ${CMAKE_COMMAND}
                -DMSL_SOURCE=${source}
                -DMSL_OUTPUT=${output}
                -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed_msl.cmake
        DEPENDS ${source} tessera_shader_cross_${shader_id}
        COMMENT "embed MSL ${shader_id}.${stage}"
        VERBATIM)
    set(${out_var} ${output} PARENT_SCOPE)
endfunction()

set(TESSERA_METAL_MSL_SOURCES)
set(TESSERA_METAL_HAS_MSL FALSE)
if (TESSERA_SHADER_CROSS)
    tessera_embed_msl(batched2d vert TESSERA_MSL_BATCHED2D_VERT)
    tessera_embed_msl(batched2d frag TESSERA_MSL_BATCHED2D_FRAG)
    tessera_embed_msl(glass_blur vert TESSERA_MSL_GLASS_BLUR_VERT)
    tessera_embed_msl(glass_blur frag TESSERA_MSL_GLASS_BLUR_FRAG)
    set(TESSERA_METAL_MSL_SOURCES
        ${TESSERA_MSL_BATCHED2D_VERT}
        ${TESSERA_MSL_BATCHED2D_FRAG}
        ${TESSERA_MSL_GLASS_BLUR_VERT}
        ${TESSERA_MSL_GLASS_BLUR_FRAG})
    set(TESSERA_METAL_HAS_MSL TRUE)
    message(STATUS "Metal: MSL shader table embedded from ${TESSERA_METAL_MSL_DIR}")
else()
    message(STATUS
        "Metal: TESSERA_SHADER_CROSS is OFF, so the MSL shader table is empty; a Metal build needs "
        "-DTESSERA_SHADER_CROSS=ON (ADR-053)")
endif()

add_library(tessera_metal_mapping STATIC
    src/metal_mapping.cpp
    src/metal_shader_registry.cpp
    ${TESSERA_METAL_MSL_SOURCES})

target_include_directories(tessera_metal_mapping
    PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    PRIVATE ${TESSERA_METAL_MSL_GEN_DIR})
target_link_libraries(tessera_metal_mapping PUBLIC tessera_rhi)
target_compile_features(tessera_metal_mapping PUBLIC cxx_std_23)

if (TESSERA_METAL_HAS_MSL)
    target_compile_definitions(tessera_metal_mapping PRIVATE TESSERA_METAL_HAS_MSL=1)
    # The MSL files are build artifacts of the cross step, so the library must not compile before
    # they exist.
    add_dependencies(tessera_metal_mapping
        tessera_shader_cross_batched2d
        tessera_shader_cross_glass_blur)
endif()

if (TESSERA_BUILD_METAL)
    if (NOT APPLE)
        message(FATAL_ERROR
            "TESSERA_BUILD_METAL=ON needs an Apple platform: the backend is Objective-C++ "
            "against <Metal/Metal.h> (ADR-053).")
    endif()

    enable_language(OBJCXX)

    # The device half. Objective-C++ (ARC) is confined to this directory: no `id<...>` or `NS*`
    # type may appear in a header outside it (docs/design/gpu/metal-backend.md section 4).
    add_library(tessera_rhi_metal STATIC
        src/metal_device.mm
        src/metal_command_list.mm
        src/metal_pipeline.mm
        src/metal_swapchain.mm)

    target_include_directories(tessera_rhi_metal
        PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
    target_link_libraries(tessera_rhi_metal PUBLIC tessera_metal_mapping tessera_rhi_shared)
    target_compile_features(tessera_rhi_metal PUBLIC cxx_std_23)

    # ARC for the Objective-C++ translation units only; the C++ headers stay plain C++.
    set_source_files_properties(
        src/metal_device.mm
        src/metal_command_list.mm
        src/metal_pipeline.mm
        src/metal_swapchain.mm
        PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    target_compile_options(tessera_rhi_metal PRIVATE "-fobjc-arc")

    # Metal, Foundation and QuartzCore (CAMetalLayer) are system frameworks: dynamic, as ADR-012
    # requires for OS components.
    target_link_libraries(tessera_rhi_metal PRIVATE
        "-framework Metal"
        "-framework Foundation"
        "-framework QuartzCore")
endif()
